// TownScout API proxy — deploy this on Cloudflare Workers (free tier). // Holds your Anthropic and Google API keys as secrets and proxies requests to them, // so neither key is ever exposed in the browser. // // Required secrets (Settings -> Variables and Secrets): // ANTHROPIC_API_KEY // GOOGLE_PLACES_API_KEY const ALLOWED_ORIGINS = [ 'https://thetownscout.com', 'https://www.thetownscout.com', 'https://kaplutus.github.io' ]; function corsHeaders(origin) { const allow = ALLOWED_ORIGINS.includes(origin) ? origin : ALLOWED_ORIGINS[0]; return { 'Access-Control-Allow-Origin': allow, 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type' }; } export default { async fetch(request, env) { const origin = request.headers.get('Origin') || ''; const url = new URL(request.url); if (request.method === 'OPTIONS') { return new Response(null, { headers: corsHeaders(origin) }); } if (!ALLOWED_ORIGINS.includes(origin)) { return new Response('Forbidden', { status: 403, headers: corsHeaders(origin) }); } try { if (url.pathname === '/anthropic' && request.method === 'POST') { return await proxyAnthropic(request, env, origin); } if (url.pathname === '/geocode' && request.method === 'GET') { return await proxyGeocode(url, env, origin); } if (url.pathname === '/reverse-geocode' && request.method === 'GET') { return await proxyReverseGeocode(url, env, origin); } if (url.pathname === '/places-nearby' && request.method === 'POST') { return await proxyPlacesNearby(request, env, origin); } if (url.pathname === '/photo' && request.method === 'GET') { return await proxyPhoto(url, env, origin); } return new Response('Not found', { status: 404, headers: corsHeaders(origin) }); } catch (err) { return new Response(JSON.stringify({ error: String(err && err.message || err) }), { status: 500, headers: { 'Content-Type': 'application/json', ...corsHeaders(origin) } }); } } }; async function proxyAnthropic(request, env, origin) { const body = await request.text(); const upstream = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': env.ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01' }, body }); const data = await upstream.text(); return new Response(data, { status: upstream.status, headers: { 'Content-Type': 'application/json', ...corsHeaders(origin) } }); } async function proxyGeocode(url, env, origin) { const zip = url.searchParams.get('zip'); if (!zip) return new Response(JSON.stringify({ error: 'missing zip' }), { status: 400, headers: corsHeaders(origin) }); const gUrl = `https://maps.googleapis.com/maps/api/geocode/json?address=${encodeURIComponent(zip)}&key=${env.GOOGLE_PLACES_API_KEY}`; const upstream = await fetch(gUrl); const data = await upstream.text(); return new Response(data, { status: upstream.status, headers: { 'Content-Type': 'application/json', ...corsHeaders(origin) } }); } async function proxyReverseGeocode(url, env, origin) { const lat = url.searchParams.get('lat'); const lng = url.searchParams.get('lng'); if (!lat || !lng) return new Response(JSON.stringify({ error: 'missing lat/lng' }), { status: 400, headers: corsHeaders(origin) }); const gUrl = `https://maps.googleapis.com/maps/api/geocode/json?latlng=${lat},${lng}&key=${env.GOOGLE_PLACES_API_KEY}`; const upstream = await fetch(gUrl); const data = await upstream.text(); return new Response(data, { status: upstream.status, headers: { 'Content-Type': 'application/json', ...corsHeaders(origin) } }); } async function proxyPlacesNearby(request, env, origin) { const body = await request.text(); const upstream = await fetch('https://places.googleapis.com/v1/places:searchNearby', { method: 'POST', headers: { 'Content-Type': 'application/json', 'X-Goog-Api-Key': env.GOOGLE_PLACES_API_KEY, 'X-Goog-FieldMask': [ 'places.id', 'places.displayName', 'places.formattedAddress', 'places.location', 'places.rating', 'places.priceLevel', 'places.types', 'places.primaryType', 'places.currentOpeningHours.openNow', 'places.businessStatus', 'places.photos', 'places.editorialSummary' ].join(',') }, body }); const data = await upstream.text(); return new Response(data, { status: upstream.status, headers: { 'Content-Type': 'application/json', ...corsHeaders(origin) } }); } async function proxyPhoto(url, env, origin) { const name = url.searchParams.get('name'); const maxw = url.searchParams.get('maxw') || '900'; if (!name) return new Response('missing name', { status: 400, headers: corsHeaders(origin) }); const gUrl = `https://places.googleapis.com/v1/${name}/media?maxWidthPx=${encodeURIComponent(maxw)}&key=${env.GOOGLE_PLACES_API_KEY}`; const upstream = await fetch(gUrl, { redirect: 'follow' }); const contentType = upstream.headers.get('Content-Type') || 'image/jpeg'; return new Response(upstream.body, { status: upstream.status, headers: { 'Content-Type': contentType, 'Cache-Control': 'public, max-age=86400', ...corsHeaders(origin) } }); }
